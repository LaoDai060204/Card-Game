    public static class YgoServer
    {
        public static bool IsBusy, Reject, Running;
        public static Thread serverThread;
        public static int Starts, Stops;
        public static bool TryStartServer(string args)
        {
            Starts++;
            if (Reject) return false;
            IsBusy = Running = true;
            serverThread = new Thread(() => { });
            return true;
        }
        public static bool ServerRunning() => Running;
        public static void StopServer() { Stops++; IsBusy = Running = false; serverThread = null; }
    }
    public static class TcpHelper
    {
        public static bool canJoin, Accept, Complete;

        public static bool LinkStart(string host, string name, string port, string password, bool local, Action success)
        {
            Joins++; Success = success;

            return Accept;
        }

internal static class Fixture
{
    public static Action OnDelay;
    public static int Bots, BlackIn, BlackOut, Messages, Errors, Passed;
    private static void Require(bool value, string message) { if (!value) throw new Exception(message); }
    private static void Reset()
    {
        Program.Running = true;
        Time.realtimeSinceStartup = 0;
        OnDelay = null;
        Bots = BlackIn = BlackOut = Messages = Errors = 0;
        YgoServer.IsBusy = YgoServer.Reject = YgoServer.Running = false;
        YgoServer.Starts = YgoServer.Stops = 0; YgoServer.serverThread = null;
        TcpHelper.canJoin = TcpHelper.Accept = TcpHelper.Complete = true;
        TcpHelper.Success = null; TcpHelper.Joins = 0; TcpHelper.Packets.Clear();
        RoomServant.FromSolo = RoomServant.FromHandTest = RoomServant.FromLocalHost = false;
    }
    private static void Case(string name, Action test)
    {
        Reset(); test(); Passed++; Console.WriteLine("PASS " + name);
    }
    private static void Main()
    {
        Case("LAN creates one server and retains a successful join", () => {
            new OnlineServant().Launch();
            Require(YgoServer.Starts == 1 && YgoServer.Stops == 0 && TcpHelper.Joins == 1 && RoomServant.FromLocalHost, "Successful LAN launch was lost");
        });
        Case("LAN rejects active/stopping servers without starting or joining", () => {
            YgoServer.IsBusy = true; new OnlineServant().Launch();
            Require(YgoServer.Starts == 0 && TcpHelper.Joins == 0, "Busy server was replaced");
        });
        Case("LAN rejected native start never joins", () => {
            YgoServer.Reject = true; new OnlineServant().Launch();
            Require(YgoServer.Starts == 1 && TcpHelper.Joins == 0, "Rejected LAN native start still joined");
        });
        Case("LAN rejected connection stops only its newly created server", () => {
            TcpHelper.Accept = false; new OnlineServant().Launch();
            Require(YgoServer.Stops == 1 && !RoomServant.FromLocalHost, "Rejected LAN join leaked a server");
        });
        Case("LAN connection timeout is bounded and releases its server", () => {
            TcpHelper.Complete = false; new OnlineServant().Launch();
            Require(Time.realtimeSinceStartup >= 10 && Time.realtimeSinceStartup < 10.2f && YgoServer.Stops == 1, "LAN connection wait was unbounded");
        });
        Case("LAN Play-stop exits pending join and releases server", () => {
            TcpHelper.Complete = false; OnDelay = () => Program.Running = false; new OnlineServant().Launch();
            Require(YgoServer.Stops == 1 && Messages == 0, "Stopped Play leaked LAN state or showed stale feedback");
        });
        Case("LAN stale monitor cannot stop or reset replacement session", () => {
            TcpHelper.Complete = false;
            OnDelay = () => { YgoServer.serverThread = new Thread(() => { }); RoomServant.FromLocalHost = true; };
            new OnlineServant().Launch();
            Require(YgoServer.Stops == 0 && RoomServant.FromLocalHost && Messages == 0, "Stale LAN monitor modified replacement session");
        });
        Case("Hand-test success keeps packet order and transfers server ownership", () => {
            new MDPro3.UI.ServantUI.DeckEditorUI().Launch().GetAwaiter().GetResult();
            Require(string.Join(",", TcpHelper.Packets) == "deck,ready,hand,turn" && Bots == 1 && YgoServer.Stops == 0, "Hand-test handshake changed");
        });
        Case("Hand-test entry rejects an active native server", () => {
            YgoServer.IsBusy = true;
            Require(!new MDPro3.UI.ServantUI.DeckEditorUI().TryLaunch() && YgoServer.Starts == 0, "Busy hand test started");
        });
        Case("Hand-test rejected native start creates no AI or transition", () => {
            YgoServer.Reject = true; new MDPro3.UI.ServantUI.DeckEditorUI().Launch().GetAwaiter().GetResult();
            Require(TcpHelper.Joins == 0 && Bots == 0 && BlackIn == 0, "Rejected native start touched hand-test UI");
        });
        Case("Hand-test rejected connection cleans transition and server", () => {
            TcpHelper.Accept = false; new MDPro3.UI.ServantUI.DeckEditorUI().Launch().GetAwaiter().GetResult();
            Require(YgoServer.Stops == 1 && BlackOut == 1 && !RoomServant.FromHandTest && TcpHelper.Packets.Count == 0, "Rejected hand test leaked state");
        });
        Case("Hand-test failed connection times out without sending gameplay", () => {
            TcpHelper.Complete = false; new MDPro3.UI.ServantUI.DeckEditorUI().Launch().GetAwaiter().GetResult();
            Require(Time.realtimeSinceStartup >= 10 && Time.realtimeSinceStartup < 10.6f && YgoServer.Stops == 1 && BlackOut == 1 && TcpHelper.Packets.Count == 0, "Hand-test timeout lost cleanup");
        });
        Case("Hand-test Play-stop before AI cleans server and black transition", () => {
            OnDelay = () => Program.Running = false; new MDPro3.UI.ServantUI.DeckEditorUI().Launch().GetAwaiter().GetResult();
            Require(YgoServer.Stops == 1 && BlackOut == 1 && Bots == 0 && Messages == 0, "Play-stop spawned AI or leaked transition");
        });
        Case("Hand-test native exit stops waiting before connection deadline", () => {
            TcpHelper.Complete = false;
            OnDelay = () => { if (TcpHelper.Joins > 0) YgoServer.Running = false; };
            new MDPro3.UI.ServantUI.DeckEditorUI().Launch().GetAwaiter().GetResult();
            Require(YgoServer.Stops == 1 && Time.realtimeSinceStartup < 1 && BlackOut == 1, "Native exit left hand-test wait active");
        });
        Case("Hand-test stale continuation cannot stop replacement or clear its UI", () => {
            TcpHelper.Complete = false;
            OnDelay = () => {
                if (TcpHelper.Joins > 0) { YgoServer.serverThread = new Thread(() => { }); RoomServant.FromHandTest = true; }
            };
            new MDPro3.UI.ServantUI.DeckEditorUI().Launch().GetAwaiter().GetResult();
            Require(YgoServer.Stops == 0 && BlackOut == 0 && RoomServant.FromHandTest && Messages == 0, "Stale hand-test monitor clobbered replacement");
        });
        Case("Hand-test stops sending after Play exits between handshake actions", () => {
            OnDelay = () => { if (TcpHelper.Packets.Contains("ready")) Program.Running = false; };
            new MDPro3.UI.ServantUI.DeckEditorUI().Launch().GetAwaiter().GetResult();
            Require(string.Join(",", TcpHelper.Packets) == "deck,ready" && YgoServer.Stops == 1 && BlackOut == 1, "Late handshake packets sent after Play stopped");
        });
        Console.WriteLine("PASS: " + Passed + " production LAN/hand-test launch scenarios; controlled Unity timing, not Unity Play.");
    }
}
