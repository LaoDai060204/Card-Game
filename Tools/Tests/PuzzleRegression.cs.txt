    public static class Program
    {
        public const string PATH_PUZZLE = "Puzzle/";
        public const string EXPANSION_LUA = ".lua";
        public const string STRING_LINE_BREAK = "\r\n";
        private static int scenarios;
        private static void Require(bool condition, string message)
        {
            if (!condition) throw new Exception(message);
        }
        private static void Fixture(string directory, int count)
        {
            Directory.SetCurrentDirectory(directory);
            if (Directory.Exists("Puzzle")) Directory.Delete("Puzzle", true);
            Directory.CreateDirectory("Puzzle");
            for (int i = 0; i < count; i++)
                File.WriteAllText("Puzzle/Fixture" + i + ".lua", "Debug.AddCard(" + (100 + i) + ",0,0,0,0,0)\n");
            PuzzleSelectorUI.Reset();
            Config.Entered = Config.Cleared = false;
            Config.Last = "";
        }
        private static void CheckButtons(PuzzlePortalUI ui, int expected)
        {
            ui.Apply();
            Require(ui.RecommendButton1.gameObject.activeSelf == (expected > 0), "First recommendation state");
            Require(ui.RecommendButton2.gameObject.activeSelf == (expected > 1), "Second recommendation state");
            if (expected > 0) Require(!string.IsNullOrEmpty(ui.RecommendButton1.AssignedName), "First recommendation missing data");
            if (expected > 1) Require(!string.IsNullOrEmpty(ui.RecommendButton2.AssignedName), "Second recommendation missing data");
        }
        public static void Main(string[] args)
        {
            string root = args[0];
            string temporary = Path.Combine(Path.GetTempPath(), "puzzle-fixtures-" + Guid.NewGuid().ToString("N"));
            Directory.CreateDirectory(temporary);
            try
            {
                for (int count = 0; count <= 3; count++)
                {
                    Fixture(temporary, count);
                    var ui = new PuzzlePortalUI();
                    CheckButtons(ui, Math.Min(count, 2));
                    Require(!ui.LastPlayButton.gameObject.activeSelf, "No previous game must hide Last Play");
                    scenarios++;
                }
                var reused = new PuzzlePortalUI();
                foreach (int count in new[] { 2, 0, 1, 2 }) { Fixture(temporary, count); CheckButtons(reused, Math.Min(count, 2)); }
                scenarios++;
                Config.Last = "Puzzle/Fixture0";
                reused.Apply();
                Require(reused.LastPlayButton.gameObject.activeSelf && reused.LastPlayButton.AssignedName == "Fixture0", "Valid Last Play not bound");
                Config.Last = "Puzzle/Missing";
                reused.Apply();
                Require(!reused.LastPlayButton.gameObject.activeSelf, "Deleted Last Play remains visible");
                scenarios++;
                Config.Entered = Config.Cleared = true;
                PuzzleSelectorUI.Reset();
                Require(PuzzleSelectorUI.GetRecommendPuzzles(2).Count == 2, "Completed puzzles must remain eligible as fallback");
                scenarios++;
                var missing = PuzzleSelectorUI.GetFirstPuzzle(new PuzzleGroup { pathPrefix = "Empty/", namePattern = "" });
                Require(missing.name == null && missing.firstCard == null, "Empty group should return a safe default");
                scenarios++;

                Directory.SetCurrentDirectory(root);
                PuzzleSelectorUI.Reset();
                Config.Entered = Config.Cleared = false;
                Config.Last = "";
                using var json = JsonDocument.Parse(File.ReadAllText("Tools/RuntimeData/puzzle-source.json"));
                int imported = 0, numericFirstCards = 0;
                foreach (var record in json.RootElement.GetProperty("files").EnumerateArray())
                {
                    string path = record.GetProperty("path").GetString();
                    var puzzle = PuzzleSelectorUI.Read(path);
                    Require(puzzle.name == Path.GetFileNameWithoutExtension(path), "Name differs: " + path);
                    Require(puzzle.GetPath() == path, "Runtime path differs: " + path);
                    Require(puzzle.description != null && puzzle.solution != null, "Null puzzle metadata: " + path);
                    if (int.TryParse(puzzle.firstCard, out int card) && card > 0) numericFirstCards++;
                    imported++;
                }
                Require(imported == json.RootElement.GetProperty("fileCount").GetInt32(), "Imported count mismatch");
                scenarios++;
                foreach (var group in json.RootElement.GetProperty("folderCounts").EnumerateObject())
                {
                    string prefix = group.Name == "Puzzle" ? "" : group.Name.Substring("Puzzle/".Length) + "/";
                    int actual = PuzzleSelectorUI.Scan(new PuzzleGroup { pathPrefix = prefix, namePattern = PuzzlePortalUI.GetPuzzleNamePattern(prefix) });
                    Require(actual == group.Value.GetInt32(), "Scanner count differs: " + group.Name);
                }
                Require(PuzzleSelectorUI.GetRecommendPuzzles(200).Count == imported, "Full import is not discoverable");
                scenarios++;
                Require(PuzzleDependencies.FindMissingPuzzleCardIds(null, _ => false).Count == 0, "Null setup text must be safe");
                Require(PuzzleDependencies.FindMissingPuzzleCardIds("Debug.AddCard(100,0,0,0,0,0)", _ => true).Count == 0,
                    "Known card must not block a Puzzle");
                scenarios++;
                var missingIds = PuzzleDependencies.FindMissingPuzzleCardIds(
                    "-- Debug.AddCard(88,0,0,0,0,0)\n" +
                    "Debug.AddCard(30,0,0,0,0,0)\n" +
                    "local a = Debug.AddCard(20,0,0,0,0,0)\n" +
                    "b = Debug.AddCard(30,0,0,0,0,0)\n" +
                    "Debug.AddCard(100,0,0,0,0,0)\n" +
                    "Debug.AddCard(cardId,0,0,0,0,0)\n" +
                    "Debug.AddCard(0,0,0,0,0,0)\n" +
                    "Debug.AddCard(9999999999999999999999,0,0,0,0,0)\n" +
                    "description = 'unrelated 77'\n", id => id == 100);
                Require(missingIds.SequenceEqual(new[] { 20, 30 }), "Missing IDs must be sorted, unique and limited to literal setup calls");
                scenarios++;
                bool rejectedNullPredicate = false;
                try { PuzzleDependencies.FindMissingPuzzleCardIds("", null); }
                catch (ArgumentNullException) { rejectedNullPredicate = true; }
                Require(rejectedNullPredicate, "A missing lookup predicate must not silently mark every card as absent");
                scenarios++;
                var known = new HashSet<int>(JsonSerializer.Deserialize<int[]>(File.ReadAllText(args[1])));
                int incompletePuzzles = 0;
                var legacy = new HashSet<int>();
                foreach (var record in json.RootElement.GetProperty("files").EnumerateArray())
                {
                    string path = record.GetProperty("path").GetString();
                    var missingCards = PuzzleDependencies.FindMissingPuzzleCardIds(File.ReadAllText(path), known.Contains);
                    if (missingCards.Count > 0) incompletePuzzles++;
                    foreach (int id in missingCards) legacy.Add(id);
                }
                Require(incompletePuzzles == 5 && imported - incompletePuzzles == 99, "Unexpected legacy dependency coverage");
                Require(legacy.SetEquals(new[] { 511000228, 511000229, 511001039, 511002993, 511002995, 511002997 }),
                    "Wrong legacy card IDs reported by the production preflight helper");
                scenarios++;
                Console.WriteLine(JsonSerializer.Serialize(new { status = "PASS", scenarios, actualProductionMethods = 12,
                    importedPuzzlesParsed = imported, numericFirstCards, puzzlesWithCompleteBaseCardData = imported - incompletePuzzles,
                    puzzlesNeedingLegacyCards = incompletePuzzles, missingLegacyCardIds = legacy.OrderBy(id => id), unityPlayTested = false }));
            }
            finally
            {
                Directory.SetCurrentDirectory(root);
                Directory.Delete(temporary, true);
            }
        }
    }
